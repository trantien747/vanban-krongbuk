Ghi ngược về Google Sheets

Khi phòng ban đánh dấu Hoàn tất, kết quả sẽ tự ghi vào cột Ghi chú trên Sheet.
Mày cần thêm đoạn code sau vào Google Apps Script (cùng file AppScript hiện tại):
// Thêm vào cuối file AppScript của mày function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.openById('1A5i0OhOuoj_1SXprESA2_5HUA2D2vkXManeyyRSL9WI').getSheetByName('Theodoi'); var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (values[i][0] == data.so_ky_hieu) { if (data.action === 'update_phong') { // Cập nhật cột D (phòng xử lý) sheet.getRange(i + 1, 4).setValue(data.phong_xu_ly); } else { // Mặc định: cập nhật cột G (ghi chú) sheet.getRange(i + 1, 7).setValue(data.ghi_chu); } break; } } return ContentService.createTextOutput(JSON.stringify({status:'ok'})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status:'error',msg:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
